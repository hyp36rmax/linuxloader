cmake_minimum_required(VERSION 3.24)
project(experience_runtime_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../.. OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE)
add_executable(experience_runtime_tests runtime_tests.cpp runtime_log_stub.c ../../src/loader/config/experienceRuntime.cpp ../../src/loader/config/iniParser.c ../../src/loader/hardware/lindbergh/eepromSettings.c)
target_compile_definitions(experience_runtime_tests PRIVATE EXPERIENCE_BUILD_REVISION="${REV}")
if(WIN32)
  target_link_libraries(experience_runtime_tests PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME effective_paths_and_saves COMMAND experience_runtime_tests)
set_tests_properties(effective_paths_and_saves PROPERTIES TIMEOUT 30)
